    99→function CentralStats() {
   100→  const { orders } = useAllOrders()
   101→  const today = new Date().toDateString()
   102→
   103→  const stats = useMemo(() => {
   104→    const todayOrders = orders.filter((o) => new Date(o.requestedAt).toDateString() === today)
   105→    const pending = orders.filter((o) => o.status === 'pendente')
   106→    const inTransit = orders.filter((o) => o.status === 'em_transito')
   107→    const deliveredToday = todayOrders.filter((o) => o.status === 'entregue')
   108→    const revenueToday = deliveredToday.reduce((s, o) => s + o.totalKz, 0)
   109→    const durations = deliveredToday
   110→      .filter((o) => o.deliveredAt)
   111→      .map((o) => new Date(o.deliveredAt!).getTime() - new Date(o.requestedAt).getTime())
   112→    const avgMin = durations.length
   113→      ? Math.round((durations.reduce((s, d) => s + d, 0) / durations.length) / 60000)
   114→      : null
   115→    return { todayCount: todayOrders.length, pending: pending.length, inTransit: inTransit.length, deliveredToday: deliveredToday.length, revenueToday, avgMin }
   116→  }, [orders, today])
   117→
   118→  const items = [
   119→    { icon: ClipboardList, label: 'pedidos hoje', value: stats.todayCount, accent: 'text-primary bg-secondary' },
   120→    { icon: Timer, label: 'solicitados', value: stats.pending, accent: 'text-amber-600 bg-amber-100' },
   121→    { icon: Truck, label: 'em trânsito', value: stats.inTransit, accent: 'text-blue-600 bg-blue-100' },
   122→    { icon: Check, label: 'entregues hoje', value: stats.deliveredToday, accent: 'text-emerald-600 bg-emerald-100' },
   123→    { icon: TrendingUp, label: 'faturação hoje', value: formatKz(stats.revenueToday), accent: 'text-gold bg-gold/15' },
   124→    { icon: Bike, label: 'tempo médio', value: stats.avgMin != null ? `${stats.avgMin} min` : '—', accent: 'text-navy bg-secondary' },
   125→  ]
   126→
   127→  return (
   128→    <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-3">
   129→      {items.map(({ icon: Icon, label, value, accent }) => (
   130→        <div key={label} className="rounded-2xl border border-border bg-white p-3 shadow-md">
   131→          <span className={cn('flex h-8 w-8 items-center justify-center rounded-lg', accent)}>
   132→            <Icon className="h-4 w-4" />
   133→          </span>
   134→          <p className="mt-2 truncate text-lg font-black leading-none text-navy">{value}</p>
   135→          <p className="mt-1 text-[10px] font-bold uppercase tracking-wide text-muted-foreground">{label}</p>
   136→        </div>
   137→      ))}
   138→    </div>
   139→  )
   140→}

   223→function CentralOrderCard({ order, onOpen }: { order: Order; onOpen: () => void }) {
   224→  const { couriers } = useCouriers()
   225→  const { toast } = useToast()
   226→  const [busy, setBusy] = useState(false)
   227→  const [assignOpen, setAssignOpen] = useState(false)
   228→
   229→  async function act(action: 'confirm' | 'assign' | 'cancel', courierId?: string) {
   230→    setBusy(true)
   231→    try {
   232→      await orderAction(order.id, action, courierId)
   233→      if (action === 'confirm') toast({ title: `Pedido nº ${order.number} confirmado ✅` })
   234→      if (action === 'assign') toast({ title: `Motoboy atribuído ao pedido nº ${order.number} 🛵` })
   235→      if (action === 'cancel') toast({ title: `Pedido nº ${order.number} cancelado`, variant: 'destructive' })
   236→    } catch (err) {
   237→      toast({
   238→        title: 'Acção falhou',
   239→        description: err instanceof Error ? err.message : 'Tente novamente.',
   240→        variant: 'destructive',
   241→      })
   242→    } finally {
   243→      setBusy(false)
   244→      setAssignOpen(false)
   245→    }
   246→  }
   247→
   248→  const availableCouriers = couriers
   249→
   250→  return (
   251→    <div className="rounded-2xl border border-border bg-white p-4 shadow-sm">
   252→      <button onClick={onOpen} className="w-full text-left" aria-label={`Detalhe do pedido nº ${order.number}`}>
   253→        <div className="flex items-start justify-between gap-2">
   254→          <div>
   255→            <p className="text-sm font-black text-navy">

   281→        {order.status === 'pendente' && (
   282→          <Button size="sm" className="h-8 rounded-full font-bold" disabled={busy} onClick={() => act('confirm')}>
   283→            {busy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Check className="h-3.5 w-3.5" />} Confirmar
   284→          </Button>

   622→function ProductFormDialog({
   623→  open,
   624→  onOpenChange,
   625→  product,
   626→  onSaved,
   627→}: {
   628→  open: boolean
   629→  onOpenChange: (o: boolean) => void
   630→  product?: Product
   631→  onSaved: () => void
   632→}) {
   633→  const { toast } = useToast()
   634→  const [name, setName] = useState(product?.name ?? '')
   635→  const [price, setPrice] = useState(String(product?.priceKz ?? ''))
   636→  const [stock, setStock] = useState(String(product?.stock ?? '50'))
   637→  const [category, setCategory] = useState(product?.category ?? 'agua')
   638→  const [volume, setVolume] = useState(String(product?.volumeL ?? ''))
   639→  const [description, setDescription] = useState(product?.description ?? '')
   640→  // FOTO REAL do artigo (data URL base64)
   641→  const [image, setImage] = useState<string | null>(product?.image ?? null)
   642→  const [imgBusy, setImgBusy] = useState(false)
   643→  const [saving, setSaving] = useState(false)
   644→
   645→  const fileInputId = `pf-photo-${product?.id ?? 'novo'}`
   646→  const cameraInputId = `pf-camera-${product?.id ?? 'novo'}`
   647→
   648→  async function onPickPhoto(file: File | undefined) {
   649→    if (!file) return
   650→    setImgBusy(true)
   651→    try {
   652→      const dataUrl = await resizeImageToDataUrl(file, 900, 0.82)
   653→      setImage(dataUrl)
   654→      toast({
   655→        title: 'Foto carregada ✅',
   656→        description: 'Não esqueça de guardar para publicar na loja.',
   657→      })
   658→    } catch (err) {
   659→      toast({
   660→        title: 'Foto inválida',
   661→        description: err instanceof Error ? err.message : 'Escolha outra imagem.',
   662→        variant: 'destructive',
   663→      })
   664→    } finally {
   665→      setImgBusy(false)
   666→    }
   667→  }
   668→
   669→  async function save() {
   670→    if (!name.trim() || !price) {
   671→      toast({ title: 'Nome e preço são obrigatórios', variant: 'destructive' })
   672→      return
   673→    }
   674→    setSaving(true)
   675→    try {
   676→      const body = {
   677→        name: name.trim(),
   678→        description: description.trim(),
   679→        priceKz: Number(price),
   680→        stock: Number(stock) || 0,
   681→        category,
   682→        kind: category === 'agua' ? (Number(volume) >= 5 ? 'garrafao' : 'garrafa') : 'acessorio',
   683→        volumeL: Number(volume) || 0,
   684→        unitLabel: category === 'agua' ? `${volume || ''}L`.trim() : 'unidade',
   685→        image, // null remove a foto, data URL guarda a foto real
   686→      }
   687→      if (product) {
   688→        await updateProduct(product.id, body)
   689→        toast({ title: 'Produto actualizado ✅', description: 'A loja do cliente já mostra as alterações.' })
   690→      } else {
   691→        await createProduct(body)
   692→        toast({ title: 'Produto criado e à venda! ✅', description: 'Já está visível na loja do cliente.' })
   693→      }
   694→      onSaved()
   695→    } catch (err) {
   696→      toast({
   697→        title: 'Erro ao guardar',
   698→        description: err instanceof Error ? err.message : 'Tente novamente.',
   699→        variant: 'destructive',
   700→      })
   701→    } finally {
   702→      setSaving(false)
   703→    }
   704→  }
   705→
   706→  return (
   707→    <Dialog open={open} onOpenChange={onOpenChange}>
   708→      <DialogContent className="max-h-[90dvh] overflow-y-auto sm:max-w-md">
   709→        <DialogHeader>
   710→          <DialogTitle>{product ? 'Editar artigo da loja' : 'Novo artigo da loja'}</DialogTitle>
   711→          <DialogDescription>
   712→            Artigos activos ficam disponíveis na <b>loja do cliente</b> assim que guardar.
   713→          </DialogDescription>
   714→        </DialogHeader>
   715→        <div className="space-y-4">
   716→          {/* ---------- FOTO REAL DO ARTIGO ---------- */}
   717→          <div className="space-y-2">
   718→            <Label className="text-sm font-black text-navy">
   719→              <Camera className="mr-1 inline h-4 w-4 text-primary" /> FOTO REAL do artigo
   720→            </Label>
   721→            {image ? (
   722→              <div className="relative overflow-hidden rounded-2xl border-2 border-primary/30 bg-secondary/50">
   723→                <img src={image} alt="Pré-visualização da foto real do artigo" className="mx-auto max-h-44 w-auto object-contain" />
   724→                <button
   725→                  type="button"
   726→                  onClick={() => setImage(null)}
   727→                  className="absolute right-2 top-2 flex h-9 w-9 items-center justify-center rounded-full bg-red-500 text-white shadow-md transition-all hover:bg-red-600 active:scale-90"
   728→                  aria-label="Remover foto"
   729→                >
   730→                  <Trash2 className="h-4 w-4" />
   731→                </button>
   732→                <span className="absolute bottom-2 left-2 rounded-full bg-navy/85 px-2.5 py-1 text-[10px] font-black text-white">
   733→                  foto real — visível na loja do cliente
   734→                </span>
   735→              </div>
   736→            ) : (
   737→              <label
   738→                htmlFor={fileInputId}
   739→                className="flex min-h-28 cursor-pointer flex-col items-center justify-center gap-1.5 rounded-2xl border-2 border-dashed border-primary/40 bg-secondary/40 p-4 text-center transition-all hover:border-primary hover:bg-secondary/70"
   740→              >
   741→                {imgBusy ? (
   742→                  <Loader2 className="h-7 w-7 animate-spin text-primary" />
   743→                ) : (
   744→                  <ImagePlus className="h-7 w-7 text-primary" />
   745→                )}
   746→                <span className="text-sm font-black text-navy">Toque para escolher a foto</span>
   747→                <span className="text-[11px] leading-snug text-muted-foreground">
   748→                  A foto real do artigo fica visível para o cliente na loja
   749→                </span>
   750→              </label>
   751→            )}
   752→            <input
   753→              id={fileInputId}
   754→              type="file"
   755→              accept="image/*"
   756→              className="hidden"
   757→              onChange={(e) => {
   758→                onPickPhoto(e.target.files?.[0])

   762→            {/* câmara directa (telemóvel) */}

   764→              id={cameraInputId}
   765→              type="file"
   766→              accept="image/*"
   767→              capture="environment"
   768→              className="hidden"
   769→              onChange={(e) => {
   770→                onPickPhoto(e.target.files?.[0])
   771→                e.target.value = ''
   772→              }}
   773→            />
   774→            <div className="grid grid-cols-2 gap-2">
   775→              <Button
   776→                type="button"
   777→                variant="outline"
   778→                className="h-10 rounded-xl text-xs font-black"
   779→                onClick={() => document.getElementById(cameraInputId)?.click()}
   780→              >
   781→                <Camera className="h-4 w-4 text-primary" /> TIRAR FOTO
   782→              </Button>
   783→              <Button
   784→                type="button"
   785→                variant="outline"
   786→                className="h-10 rounded-xl text-xs font-black"
   787→                onClick={() => document.getElementById(fileInputId)?.click()}
   788→              >
   789→                <ImagePlus className="h-4 w-4 text-primary" /> GALERIA / PC
   790→              </Button>
   791→            </div>
   792→          </div>
   793→
   794→          <div className="space-y-1.5">
   795→            <Label htmlFor="pf-name">Nome</Label>
   796→            <Input id="pf-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Ex.: Bidão 5L" />
   797→          </div>
   798→
   799→          {/* ---------- PREÇO em destaque ---------- */}

    93→function CentralStats() {
    94→  const { orders } = useAllOrders()
    95→  const today = new Date().toDateString()
    96→
    97→  const stats = useMemo(() => {
    98→    const todayOrders = orders.filter((o) => new Date(o.requestedAt).toDateString() === today)
    99→    const pending = orders.filter((o) => o.status === 'pendente')
   100→    const inTransit = orders.filter((o) => o.status === 'em_transito')
   101→    const deliveredToday = todayOrders.filter((o) => o.status === 'entregue')
   102→    const revenueToday = deliveredToday.reduce((s, o) => s + o.totalKz, 0)
   103→    const durations = deliveredToday
   104→      .filter((o) => o.deliveredAt)
   105→      .map((o) => new Date(o.deliveredAt!).getTime() - new Date(o.requestedAt).getTime())
   106→    const avgMin = durations.length
   107→      ? Math.round((durations.reduce((s, d) => s + d, 0) / durations.length) / 60000)
   108→      : null
   109→    return { todayCount: todayOrders.length, pending: pending.length, inTransit: inTransit.length, deliveredToday: deliveredToday.length, revenueToday, avgMin }
   110→  }, [orders, today])
   111→
   112→  const items = [
   113→    { icon: ClipboardList, label: 'pedidos hoje', value: stats.todayCount, accent: 'text-primary bg-secondary' },
   114→    { icon: Timer, label: 'solicitados', value: stats.pending, accent: 'text-amber-600 bg-amber-100' },
   115→    { icon: Truck, label: 'em trânsito', value: stats.inTransit, accent: 'text-blue-600 bg-blue-100' },
   116→    { icon: Check, label: 'entregues hoje', value: stats.deliveredToday, accent: 'text-emerald-600 bg-emerald-100' },
   117→    { icon: TrendingUp, label: 'faturação hoje', value: formatKz(stats.revenueToday), accent: 'text-gold bg-gold/15' },
   118→    { icon: Bike, label: 'tempo médio', value: stats.avgMin != null ? `${stats.avgMin} min` : '—', accent: 'text-navy bg-secondary' },
   119→  ]
   120→
   121→  return (
   122→    <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-3">
   123→      {items.map(({ icon: Icon, label, value, accent }) => (
   124→        <div key={label} className="rounded-2xl border border-border bg-white p-3 shadow-md">
   125→          <span className={cn('flex h-8 w-8 items-center justify-center rounded-lg', accent)}>
   126→            <Icon className="h-4 w-4" />
   127→          </span>
   128→          <p className="mt-2 truncate text-lg font-black leading-none text-navy">{value}</p>
   129→          <p className="mt-1 text-[10px] font-bold uppercase tracking-wide text-muted-foreground">{label}</p>
   130→        </div>
   131→      ))}
   132→    </div>
   133→  )
   134→}

   217→function CentralOrderCard({ order, onOpen }: { order: Order; onOpen: () => void }) {
   218→  const { couriers } = useCouriers()
   219→  const { toast } = useToast()
   220→  const [busy, setBusy] = useState(false)
   221→  const [assignOpen, setAssignOpen] = useState(false)
   222→
   223→  async function act(action: 'confirm' | 'assign' | 'cancel', courierId?: string) {
   224→    setBusy(true)
   225→    try {
   226→      await orderAction(order.id, action, courierId)
   227→      if (action === 'confirm') toast({ title: `Pedido nº ${order.number} confirmado ✅` })
   228→      if (action === 'assign') toast({ title: `Motoboy atribuído ao pedido nº ${order.number} 🛵` })
   229→      if (action === 'cancel') toast({ title: `Pedido nº ${order.number} cancelado`, variant: 'destructive' })
   230→    } catch (err) {
   231→      toast({
   232→        title: 'Acção falhou',
   233→        description: err instanceof Error ? err.message : 'Tente novamente.',
   234→        variant: 'destructive',
   235→      })
   236→    } finally {
   237→      setBusy(false)
   238→      setAssignOpen(false)
   239→    }
   240→  }
   241→
   242→  const availableCouriers = couriers
   243→
   244→  return (
   245→    <div className="rounded-2xl border border-border bg-white p-4 shadow-sm">
   246→      <button onClick={onOpen} className="w-full text-left" aria-label={`Detalhe do pedido nº ${order.number}`}>
   247→        <div className="flex items-start justify-between gap-2">
   248→          <div>
   249→            <p className="text-sm font-black text-navy">

   262→            <b className="text-foreground">{formatKz(order.totalKz)}</b>

   274→      <div className="mt-3 flex flex-wrap gap-2 border-t border-border pt-3">
   275→        {order.status === 'pendente' && (
   276→          <Button size="sm" className="h-8 rounded-full font-bold" disabled={busy} onClick={() => act('confirm')}>
   277→            {busy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Check className="h-3.5 w-3.5" />} Confirmar
   278→          </Button>

   308→                key={c.id}

   460→    <div className="space-y-3">

   462→        <div key={f.id} className="rounded-2xl border border-border bg-white p-4 shadow-sm">
   463→          <div className="flex items-start justify-between gap-2">

   590→function ProductFormDialog({
   591→  open,
   592→  onOpenChange,
   593→  product,
   594→  onSaved,
   595→}: {
   596→  open: boolean
   597→  onOpenChange: (o: boolean) => void
   598→  product?: Product
   599→  onSaved: () => void
   600→}) {
   601→  const { toast } = useToast()
   602→  const [name, setName] = useState(product?.name ?? '')
   603→  const [price, setPrice] = useState(String(product?.priceKz ?? ''))
   604→  const [stock, setStock] = useState(String(product?.stock ?? '50'))
   605→  const [category, setCategory] = useState(product?.category ?? 'agua')
   606→  const [volume, setVolume] = useState(String(product?.volumeL ?? ''))
   607→  const [saving, setSaving] = useState(false)
   608→
   609→  async function save() {
   610→    if (!name.trim() || !price) {
   611→      toast({ title: 'Nome e preço são obrigatórios', variant: 'destructive' })
   612→      return
   613→    }
   614→    setSaving(true)
   615→    try {
   616→      const body = {
   617→        name: name.trim(),
   618→        priceKz: Number(price),
   619→        stock: Number(stock) || 0,
   620→        category,
   621→        kind: category === 'agua' ? (Number(volume) >= 5 ? 'garrafao' : 'garrafa') : 'acessorio',
   622→        volumeL: Number(volume) || 0,
   623→        unitLabel: category === 'agua' ? `${volume ?? ''}L` : 'unidade',
   624→      }
   625→      if (product) {
   626→        await updateProduct(product.id, body)
   627→        toast({ title: 'Produto actualizado ✅' })
   628→      } else {
   629→        await createProduct(body)
   630→        toast({ title: 'Produto criado e à venda! ✅' })
   631→      }
   632→      onSaved()
   633→    } catch (err) {
   634→      toast({
   635→        title: 'Erro ao guardar',
   636→        description: err instanceof Error ? err.message : 'Tente novamente.',
   637→        variant: 'destructive',
   638→      })
   639→    } finally {
   640→      setSaving(false)
   641→    }
   642→  }
   643→
   644→  return (
   645→    <Dialog open={open} onOpenChange={onOpenChange}>
   646→      <DialogContent className="sm:max-w-sm">
   647→        <DialogHeader>
   648→          <DialogTitle>{product ? 'Editar produto' : 'Novo produto'}</DialogTitle>
   649→          <DialogDescription>
   650→            Produtos activos ficam disponíveis na loja do cliente.
   651→          </DialogDescription>
   652→        </DialogHeader>
   653→        <div className="space-y-3">
   654→          <div className="space-y-1.5">
   655→            <Label htmlFor="pf-name">Nome</Label>
   656→            <Input id="pf-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Ex.: Bidão 5L" />
   657→          </div>
   658→          <div className="grid grid-cols-2 gap-2">
   659→            <div className="space-y-1.5">
   660→              <Label htmlFor="pf-price">Preço (Kz)</Label>
   661→              <Input id="pf-price" inputMode="numeric" value={price} onChange={(e) => setPrice(e.target.value)} placeholder="600" />
   662→            </div>
   663→            <div className="space-y-1.5">
   664→              <Label htmlFor="pf-stock">Stock</Label>
   665→              <Input id="pf-stock" inputMode="numeric" value={stock} onChange={(e) => setStock(e.target.value)} placeholder="100" />
   666→            </div>
   667→          </div>
   668→          <div className="grid grid-cols-2 gap-2">
   669→            <div className="space-y-1.5">
   670→              <Label>Categoria</Label>
   671→              <Select value={category} onValueChange={setCategory}>
   672→                <SelectTrigger><SelectValue /></SelectTrigger>
   673→                <SelectContent>
   674→                  <SelectItem value="agua">Água</SelectItem>
   675→                  <SelectItem value="acessorio">Acessório</SelectItem>
   676→                </SelectContent>
   677→              </Select>
   678→            </div>
   679→            <div className="space-y-1.5">
   680→              <Label htmlFor="pf-vol">Volume (L)</Label>
   681→              <Input id="pf-vol" inputMode="decimal" value={volume} onChange={(e) => setVolume(e.target.value)} placeholder="5" disabled={category === 'acessorio'} />
   682→            </div>
   683→          </div>
   684→        </div>
   685→        <DialogFooter>
   686→          <Button onClick={save} disabled={saving} className="w-full font-bold">
   687→            {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <Check className="h-4 w-4" />}
   688→            Guardar produto
   689→          </Button>
   690→        </DialogFooter>
   691→      </DialogContent>
   692→    </Dialog>
   693→  )
   694→}
